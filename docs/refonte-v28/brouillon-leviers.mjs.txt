// Brouillon jetable n° 2 : où vit l'habileté ? Le choix du trio seul, le contre d'un rôle, le tempo (fermer / ouvrir).
// Un coach « par essais » (il rejoue la fin du match `ROLL` fois pour chaque geste, sur les mêmes dés) donne le plafond de ce qu'un bon joueur peut tirer d'un levier.
let seed = 777;
const rnd = () => (seed = (seed * 1664525 + 1013904223) >>> 0) / 4294967296;
const poisson = l => { let L = Math.exp(-l), k = 0, p = 1; do { k++; p *= rnd(); } while (p > L); return k - 1; };
const ENV = JSON.parse(process.env.P || '{}');
const P = { PRES: 5, T0: 2.0, K: 1.5, P0: 0.095, COUT: 30, REPOS: 15, ENTRACTE: 20, ZONE: 0.25, JMIN: 0.7, CHECK: 0, TEMPO: 0, ROLL: 60, ...ENV };
const club = (d = 0) => ({
  trios: [ { att: 16 + d, def: 11 + d, tir: 17 + d, rob: 10 }, { att: 14 + d, def: 12 + d, tir: 14 + d, rob: 11 }, { att: 11 + d, def: 15 + d, tir: 11 + d, rob: 13, check: 1 }, { att: 10 + d, def: 11 + d, tir: 10 + d, rob: 16 } ],
  paires: [ { att: 13 + d, def: 15 + d }, { att: 12 + d, def: 13 + d }, { att: 10 + d, def: 11 + d } ], g: { arr: 13 + d },
});
const j = x => P.JMIN + (1 - P.JMIN) * Math.max(0, x) / 100;
const TEMPOS = [{ pour: 1, contre: 1 }, { pour: 0.7, contre: 0.6 }, { pour: 1.3, contre: 1.45 }]; // jouer, fermer, ouvrir
// Le checker (CHECK) : face au meilleur trio adverse (le 1er), il lui retire une part de son attaque et de son tir.
const lam = (E, t, p, O, to, po, zone, tp, tpo) => {
  const a = E.c.trios[t], b = O.c.trios[to];
  const frein = (b.check && t === 0) ? 1 - P.CHECK : 1;
  const att = (0.7 * a.att * frein + 0.3 * E.c.paires[p].att) * j(E.jt[t]), def = (0.4 * b.def + 0.6 * O.c.paires[po].def) * j(O.jt[to]);
  return P.T0 * Math.pow(att / def, P.K) * (1 + P.ZONE * zone) * TEMPOS[tp].pour * TEMPOS[tpo].contre;
};
const pBut = (E, t, O, to) => Math.min(0.35, P.P0 * (E.c.trios[t].tir * ((O.c.trios[to].check && t === 0) ? 1 - P.CHECK / 2 : 1) / O.c.g.arr));
const neuf = (cA, cB) => ({ A: { c: cA, jt: [100, 100, 100, 100], jp: [100, 100, 100], buts: 0 }, B: { c: cB, jt: [100, 100, 100, 100], jp: [100, 100, 100], buts: 0 }, zone: 0, n: 0 });
const copie = S => ({ A: { c: S.A.c, jt: [...S.A.jt], jp: [...S.A.jp], buts: S.A.buts }, B: { c: S.B.c, jt: [...S.B.jt], jp: [...S.B.jp], buts: S.B.buts }, zone: S.zone, n: S.n });
const paireDe = E => E.jp.indexOf(Math.max(...E.jp));
const frais = E => E.jt.indexOf(Math.max(...E.jt));

function jouer(S, ta, tb, tpa = 0, tpb = 0, prol = false) {
  const { A, B } = S, pa = paireDe(A), pb = paireDe(B);
  const na = poisson(lam(A, ta, pa, B, tb, pb, S.zone, tpa, tpb)), nb = poisson(lam(B, tb, pb, A, ta, pa, -S.zone, tpb, tpa));
  let ga = 0, gb = 0;
  for (let i = 0; i < na; i++) if (rnd() < pBut(A, ta, B, tb)) ga++;
  for (let i = 0; i < nb; i++) if (rnd() < pBut(B, tb, A, ta)) gb++;
  if (prol && ga + gb > 0) { if (ga && gb) { rnd() < 0.5 ? gb = 0 : ga = 0; } ga = Math.min(1, ga); gb = Math.min(1, gb); }
  A.buts += ga; B.buts += gb;
  const cout = (E, O, t, to) => P.COUT + Math.max(-8, Math.min(12, O.c.trios[to].rob - E.c.trios[t].rob));
  const ca = cout(A, B, ta, tb), cb = cout(B, A, tb, ta);
  for (let i = 0; i < 4; i++) { A.jt[i] = i === ta ? A.jt[i] - ca : Math.min(100, A.jt[i] + P.REPOS); B.jt[i] = i === tb ? B.jt[i] - cb : Math.min(100, B.jt[i] + P.REPOS); }
  for (let i = 0; i < 3; i++) { A.jp[i] = i === pa ? A.jp[i] - 34 : Math.min(100, A.jp[i] + 22); B.jp[i] = i === pb ? B.jp[i] - 34 : Math.min(100, B.jp[i] + 22); }
  S.zone = ga || gb ? 0 : Math.sign(na - nb);
  S.n++;
  if (S.n % P.PRES === 0 && S.n <= 3 * P.PRES) { S.zone = 0; for (const E of [A, B]) { E.jt = E.jt.map(x => Math.min(100, x + P.ENTRACTE)); E.jp = E.jp.map(x => Math.min(100, x + P.ENTRACTE)); } }
}
// La fin du match jouée par deux coachs « frais » ; rend 1 si A gagne, 0 sinon.
function finir(S) {
  while (S.n < 3 * P.PRES) jouer(S, frais(S.A), frais(S.B));
  for (let k = 0; k < 6 && S.A.buts === S.B.buts; k++) jouer(S, frais(S.A), frais(S.B), 0, 0, true);
  return S.A.buts > S.B.buts ? 1 : S.A.buts < S.B.buts ? 0 : 0.5;
}
// Le coach par essais : pour chaque geste permis (trio × tempo), la fin du match rejouée ROLL fois.
function essais(S, cote, tb, leviers) {
  let best = null, bv = -1;
  const E = cote === 'A' ? S.A : S.B;
  const des = (seed ^ 0x9e3779b9) >>> 0;
  // Le geste par défaut d'abord : à égalité d'essais, on garde « le plus frais, jouer ».
  const ordre = [frais(E), ...[0, 1, 2, 3].filter(t => t !== frais(E))];
  for (const t of ordre) {
    if (E.jt[t] <= 0) continue;
    if (!leviers.trio && t !== frais(E)) continue;
    for (let tp = 0; tp < (leviers.tempo ? 3 : 1); tp++) {
      // Les mêmes dés pour chaque geste comparé (sinon le bruit des essais noie l'écart entre deux trios), puis le fil des dés du match reprend.
      let v = 0;
      const fil = seed; seed = des;
      for (let r = 0; r < P.ROLL; r++) { const C = copie(S); const adv = tb ?? frais(cote === 'A' ? C.B : C.A); cote === 'A' ? jouer(C, t, adv, tp, 0) : jouer(C, adv, t, 0, tp); const w = finir(C); v += cote === 'A' ? w : 1 - w; }
      seed = fil;
      if (v > bv + 1e-9) { bv = v; best = { t, tp }; }
    }
  }
  return best;
}
function match(cA, cB, polA, polB) {
  const S = neuf(cA, cB);   // A à domicile : il voit le trio de B.
  const geste = (pol, cote, vu) => pol === 'frais' ? { t: frais(cote === 'A' ? S.A : S.B), tp: 0 } : pol === 'rotation' ? { t: S.n % 4, tp: 0 } : essais(S, cote, vu, pol);
  while (S.n < 3 * P.PRES) { const b = geste(polB, 'B', null), a = geste(polA, 'A', b.t); jouer(S, a.t, b.t, a.tp, b.tp); }
  for (let k = 0; k < 6 && S.A.buts === S.B.buts; k++) { const b = geste(polB, 'B', null), a = geste(polA, 'A', b.t); jouer(S, a.t, b.t, a.tp, b.tp, true); }
  return S.A.buts > S.B.buts ? 1 : S.A.buts < S.B.buts ? 0 : (rnd() < 0.5 ? 1 : 0);
}
function duel(nom, polX, polY, N = Number(process.env.N || 1200)) {
  let w = 0;
  for (let i = 0; i < N; i++) w += i % 2 === 0 ? match(club(), club(), polX, polY) : 1 - match(club(), club(), polY, polX);
  const p = w / N, et = Math.sqrt(p * (1 - p) / N);
  console.log(`${nom.padEnd(58)} ${(100 * p).toFixed(1)} % (± ${(100 * 2 * et).toFixed(1)})`);
}
const quoi = process.env.QUOI || 'trio';
if (quoi === 'trio') duel(`le trio choisi par essais c. « le plus frais »  CHECK=${P.CHECK}`, { trio: 1 }, 'frais');
if (quoi === 'tempo') duel('le tempo seul (trio le plus frais) c. « le plus frais »', { tempo: 1 }, 'frais');
if (quoi === 'tout') duel(`trio + tempo par essais c. « le plus frais »  CHECK=${P.CHECK}`, { trio: 1, tempo: 1 }, 'frais');
if (quoi === 'rot') duel(`trio par essais c. rotation fixe  CHECK=${P.CHECK}`, { trio: 1 }, 'rotation');
