let seed = 4242;
const rnd = () => (seed = (seed * 1664525 + 1013904223) >>> 0) / 4294967296;
const poisson = l => { let L = Math.exp(-l), k = 0, p = 1; do { k++; p *= rnd(); } while (p > L); return k - 1; };
const P = { PRES: 5, T0: 2.0, K: 1.5, P0: 0.095, COUT: 30, REPOS: 15, ENTRACTE: 20, ZONE: 0.25, JMIN: 0.7 };
const base = () => ({ trios: [ { att: 16, def: 11, tir: 17 }, { att: 14, def: 12, tir: 14 }, { att: 11, def: 15, tir: 11 }, { att: 10, def: 11, tir: 10 } ],
  paires: [ { att: 13, def: 15 }, { att: 12, def: 13 }, { att: 10, def: 11 } ], g: { arr: 13 }, roul: null });
const j = x => P.JMIN + (1 - P.JMIN) * Math.max(0, x) / 100;
const lam = (E, t, p, O, to, po, zone) => P.T0 * Math.pow(((0.7 * E.c.trios[t].att + 0.3 * E.c.paires[p].att) * j(E.jt[t])) / ((0.4 * O.c.trios[to].def + 0.6 * O.c.paires[po].def) * j(O.jt[to])), P.K) * (1 + P.ZONE * zone);
const pBut = (E, t, O) => Math.min(0.4, P.P0 * E.c.trios[t].tir / O.c.g.arr);
// Le roulement : le plus frais, ou la suite écrite par le coach (`roul`, un trio par présence de la période).
const trioDe = (E, n) => { if (E.c.roul) { const t = E.c.roul[n % P.PRES]; if (E.jt[t] > 0) return t; } return E.jt.indexOf(Math.max(...E.jt)); };
function match(cA, cB) {
  const A = { c: cA, jt: [100, 100, 100, 100], jp: [100, 100, 100], buts: 0 }, B = { c: cB, jt: [100, 100, 100, 100], jp: [100, 100, 100], buts: 0 };
  let zone = 0, n = 0;
  const pres = prol => {
    const ta = trioDe(A, n), tb = trioDe(B, n), pa = A.jp.indexOf(Math.max(...A.jp)), pb = B.jp.indexOf(Math.max(...B.jp));
    const na = poisson(lam(A, ta, pa, B, tb, pb, zone)), nb = poisson(lam(B, tb, pb, A, ta, pa, -zone));
    let ga = 0, gb = 0;
    for (let i = 0; i < na; i++) if (rnd() < pBut(A, ta, B)) ga++;
    for (let i = 0; i < nb; i++) if (rnd() < pBut(B, tb, A)) gb++;
    if (prol && ga && gb) rnd() < 0.5 ? gb = 0 : ga = 0;
    A.buts += ga; B.buts += gb;
    for (let i = 0; i < 4; i++) { A.jt[i] = i === ta ? A.jt[i] - P.COUT : Math.min(100, A.jt[i] + P.REPOS); B.jt[i] = i === tb ? B.jt[i] - P.COUT : Math.min(100, B.jt[i] + P.REPOS); }
    for (let i = 0; i < 3; i++) { A.jp[i] = i === pa ? A.jp[i] - 34 : Math.min(100, A.jp[i] + 22); B.jp[i] = i === pb ? B.jp[i] - 34 : Math.min(100, B.jp[i] + 22); }
    zone = ga || gb ? 0 : Math.sign(na - nb); n++;
  };
  for (let per = 0; per < 3; per++) { zone = 0; for (let k = 0; k < P.PRES; k++) pres(false); for (const E of [A, B]) { E.jt = E.jt.map(x => Math.min(100, x + P.ENTRACTE)); E.jp = E.jp.map(x => Math.min(100, x + P.ENTRACTE)); } }
  for (let k = 0; k < 6 && A.buts === B.buts; k++) pres(true);
  return A.buts > B.buts ? 1 : A.buts < B.buts ? 0 : (rnd() < 0.5 ? 1 : 0);
}
const mesure = (nom, f, N = 40000) => { let w = 0; for (let i = 0; i < N; i++) { const X = base(); f(X); w += i % 2 ? match(X, base()) : 1 - match(base(), X); } console.log(`${nom.padEnd(64)} ${(100 * w / N).toFixed(1)} %`); };
