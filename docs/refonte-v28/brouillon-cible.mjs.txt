// La cible de l'acte : la chance de la tenir selon la part de matchs gagnés. 17 % des matchs vont en prolongation (le perdant prend 1 point).
const OT = 0.17;
function dist(p, n) { // distribution des points sur n matchs
  let d = new Map([[0, 1]]);
  for (let i = 0; i < n; i++) { const e = new Map(); for (const [pts, pr] of d) for (const [g, q] of [[2, p], [1, (1 - p) * OT], [0, (1 - p) * (1 - OT)]]) e.set(pts + g, (e.get(pts + g) || 0) + pr * q); d = e; }
  return d;
}
const tenir = (p, n, c) => [...dist(p, n)].filter(([pts]) => pts >= c).reduce((a, [, pr]) => a + pr, 0);
const serie = (p, k) => { let s = 0; const n = 2 * k - 1; const C = (a, b) => { let r = 1; for (let i = 0; i < b; i++) r = r * (a - i) / (i + 1); return r; }; for (let w = k; w <= n; w++) s += C(n, w) * p ** w * (1 - p) ** (n - w); return s; };
console.log('part gagnée | cible 6 | cible 7 | cible 8 | (sur 6 matchs)');
for (const p of [0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7, 0.75, 0.8]) console.log(`${(p * 100).toFixed(0)} %`.padEnd(11), ...[6, 7, 8].map(c => `| ${(100 * tenir(p, 6, c)).toFixed(0)} %`.padEnd(9)));
console.log('\nLa run : acte I à pI, II à pII, III à pIII, séries à pS (2 de 3, 2 de 3, 3 de 5)');
for (const [a, b, c, s] of [[0.5, 0.5, 0.5, 0.45], [0.55, 0.6, 0.6, 0.5], [0.6, 0.65, 0.65, 0.55], [0.65, 0.7, 0.7, 0.6], [0.7, 0.75, 0.8, 0.7]]) {
  const t1 = tenir(a, 6, 6), t2 = tenir(b, 6, 7), t3 = tenir(c, 6, 8);
  const vivant = 1 - (1 - t1) * (1 - t2);            // 2 cœurs : mort seulement si les deux cibles I et II sont ratées
  const series = vivant * t3;
  const coupe = series * serie(s, 2) * serie(s, 2) * serie(s, 3);
  const unCoeur = 1 - t1 * t2;
  console.log(`${a}/${b}/${c}, séries ${s} : cibles ${(100*t1).toFixed(0)}·${(100*t2).toFixed(0)}·${(100*t3).toFixed(0)} % · un cœur perdu ${(100*unCoeur).toFixed(0)} % · mort avant III ${(100*(1-vivant)).toFixed(0)} % · en séries ${(100*series).toFixed(0)} % · Coupe ${(100*coupe).toFixed(1)} %`);
}
